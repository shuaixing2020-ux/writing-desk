## 2. 条目文件的读写与 frontmatter 解析

### 数据流

```
写入：Entity 对象 → stringifyScene() → writeFileAtomic() → bible/characters/林远.md
读取：bible/*/*.md → readdir + parseFrontmatter() → listEntities() → Entity[]
```

**身份靠 frontmatter 的 `id`，不靠文件名。** 改名只改文件名的 slug 部分，id 始终不变，
所以出场索引里存的 `entity_id` 永远不会因为改名而失效。

### frontmatter 解析与序列化

这是整个文件格式的地基。**不引入 YAML 库**是刻意选择：一旦依赖第三方解析器，工具没了作者的文件就打不开。所以只支持最小的子集——扁平 key: value、`[a, b]` 数组、`|` 多行块。键名允许中文，因为设定库的自定义字段常写成「身份: 青云门弟子」。

**文件**：`server/util/frontmatter.ts`（200 行，完整）

```typescript
/**
 * 场景文件格式：Markdown 子集 + YAML 风格 frontmatter。
 *
 * 格式约束（架构 6.2）：脱离本工具也必须可读。
 * 因此只支持扁平的 key: value、数组用 [a, b]、多行文本用 | 块。
 * 不引入 YAML 库，避免"工具没了文件就打不开"。
 */

export interface Frontmatter {
  [key: string]: string | string[] | number | boolean | null | undefined;
}

const FENCE = '---';

/**
 * 解析 frontmatter + 正文。
 * 若文件不以 --- 开头，则整篇都是正文（兼容用户手工放入的 md）。
 */
export function parseFrontmatter(raw: string): { data: Frontmatter; body: string } {
  const text = raw.replace(/^\uFEFF/, ''); // 去 BOM
  if (!text.startsWith(FENCE)) {
    return { data: {}, body: text };
  }

  const lines = text.split('\n');
  // 找结束围栏
  let end = -1;
  for (let i = 1; i < lines.length; i++) {
    if (lines[i].trim() === FENCE) {
      end = i;
      break;
    }
  }
  if (end === -1) return { data: {}, body: text }; // 未闭合，按纯正文处理

  const data: Frontmatter = {};
  let i = 1;
  while (i < end) {
    const line = lines[i];
    if (!line.trim() || line.trimStart().startsWith('#')) {
      i++;
      continue;
    }
    // 键允许中文：设定库的自定义字段常直接写「身份: 青云门弟子」
    const m = /^([^\s:#][^:]*?):\s*(.*)$/.exec(line);
    if (!m) {
      i++;
      continue;
    }
    const key = m[1];
    const rest = m[2];

    // 多行块：key: |
    if (rest === '|' || rest === '|-' || rest === '>') {
      const buf: string[] = [];
      i++;
      while (i < end && (lines[i].startsWith('  ') || lines[i].trim() === '')) {
        buf.push(lines[i].replace(/^  /, ''));
        i++;
      }
      data[key] = buf.join('\n').replace(/\n+$/, '');
      continue;
    }

    data[key] = parseScalar(rest);
    i++;
  }

  // 正文从结束围栏之后开始，去掉紧跟的一个空行
  let bodyStart = end + 1;
  if (bodyStart < lines.length && lines[bodyStart].trim() === '') bodyStart++;
  // 去掉尾部空行：正文的规范化形态由 stringifyScene 统一负责，
  // 解析侧对齐它才能保证「读→写→读」幂等。
  const body = lines.slice(bodyStart).join('\n').replace(/\s+$/, '');

  return { data, body };
}

function parseScalar(raw: string): string | string[] | number | boolean | null {
  const v = raw.trim();

  if (v === '' || v === 'null' || v === '~') return null;
  if (v === 'true') return true;
  if (v === 'false') return false;

  // 数组：[a, b, c]（元素可带引号）
  if (v.startsWith('[') && v.endsWith(']')) {
    const inner = v.slice(1, -1).trim();
    if (!inner) return [];
    return splitTopLevel(inner).map(unquote);
  }

  // 数字
  if (/^-?\d+(\.\d+)?$/.test(v)) return Number(v);

  return unquote(v);
}

/** 按逗号切分，但忽略引号内的逗号 */
function splitTopLevel(s: string): string[] {
  const out: string[] = [];
  let buf = '';
  let quote: string | null = null;
  for (const ch of s) {
    if (quote) {
      buf += ch;
      if (ch === quote) quote = null;
    } else if (ch === '"' || ch === "'") {
      quote = ch;
      buf += ch;
    } else if (ch === ',') {
      out.push(buf.trim());
      buf = '';
    } else {
      buf += ch;
    }
  }
  if (buf.trim()) out.push(buf.trim());
  return out;
}

function unquote(v: string): string {
  const s = v.trim();
  if ((s.startsWith('"') && s.endsWith('"')) || (s.startsWith("'") && s.endsWith("'"))) {
    const inner = s.slice(1, -1);
    return s.startsWith('"') ? inner.replace(/\\"/g, '"').replace(/\\n/g, '\n') : inner;
  }
  return s;
}

function needsQuote(v: string): boolean {
  return (
    v === '' ||
    /^[\s]|[\s]$/.test(v) ||
    /[:#\[\]{}|>&*!%@`,]/.test(v) ||
    /^-?\d+(\.\d+)?$/.test(v) ||
    ['true', 'false', 'null', '~'].includes(v)
  );
}

function serializeScalar(v: string | string[] | number | boolean | null | undefined): string {
  if (v === null || v === undefined) return '';
  if (typeof v === 'number' || typeof v === 'boolean') return String(v);
  if (Array.isArray(v)) {
    return `[${v.map((item) => (needsQuote(item) ? JSON.stringify(item) : item)).join(', ')}]`;
  }
  if (v.includes('\n')) {
    // 多行值用块标量
    return '|\n' + v.split('\n').map((l) => '  ' + l).join('\n');
  }
  return needsQuote(v) ? JSON.stringify(v) : v;
}

/** 序列化为完整文件内容（frontmatter + 空行 + 正文） */
export function stringifyScene(data: Frontmatter, body: string): string {
  const lines: string[] = [FENCE];
  for (const [key, value] of Object.entries(data)) {
    if (value === undefined) continue;
    const ser = serializeScalar(value);
    if (ser.startsWith('|\n')) {
      lines.push(`${key}: |`);
      lines.push(ser.slice(2));
    } else if (ser === '') {
      // 空值不写，保持文件干净
      continue;
    } else {
      lines.push(`${key}: ${ser}`);
    }
  }
  lines.push(FENCE);
  lines.push('');
  // 正文保证以单个换行结尾（POSIX 文本文件约定）
  const normalized = body.replace(/\r\n/g, '\n').replace(/\s+$/, '');
  if (normalized) lines.push(normalized);
  return lines.join('\n') + '\n';
}

/** 从 frontmatter 取字符串（缺失返回默认值） */
export function fmString(data: Frontmatter, key: string, fallback = ''): string {
  const v = data[key];
  if (v === null || v === undefined) return fallback;
  if (Array.isArray(v)) return v.join(', ');
  return String(v);
}

/** 从 frontmatter 取数组 */
export function fmArray(data: Frontmatter, key: string): string[] {
  const v = data[key];
  if (Array.isArray(v)) return v;
  if (typeof v === 'string' && v.trim()) return v.split(',').map((s) => s.trim()).filter(Boolean);
  return [];
}

/** 从 frontmatter 取数字 */
export function fmNumber(data: Frontmatter, key: string, fallback = 0): number {
  const v = data[key];
  if (typeof v === 'number' && Number.isFinite(v)) return v;
  if (typeof v === 'string' && v.trim() && Number.isFinite(Number(v))) return Number(v);
  return fallback;
}
```

```typescript
            existingPath = path.join(dirPath, path.basename(existingPath));
          }

      const now = Date.now();

      const name = input.name.trim() || '未命名条目';
      const fileName = `${safeName(name)}.md`;

      const data: Frontmatter = {

        type,
        name,
        aliases: input.aliases ?? existing?.aliases ?? [],
        created_at: existing?.created_at || now,
        updated_at: now,

    });
```

## 3. 出场索引与 SQLite 建表

### 设计说明

**出场识别的方向是「给定词表找全部位置」**，和全文搜索「给定关键词找文档」正好相反。
所以它不走索引查询，而是把设定库的 name + aliases 编成模式表，逐字扫描正文。

两个必须解决的真实问题：

1. **长名优先** —「林远山」不能被拆成「林远」+「山」
2. **同名不同人** — 一个文本对应多个条目时，各自都要计数

做法：首字符分桶 + 桶内按长度降序 + 贪心前进。命中后直接跳过整段，
既不会重复计数，也不会被短名截断。条目数量在几百量级时分桶已经足够快，
不需要上 Aho–Corasick 的失败指针。

**中文搜索用的是另一套机制**（逐字索引 char_index），因为 FTS5 的 trigram
分词器对 2 字中文词（最常见的人名长度）完全失效。

```typescript
      .all(entityId) as Array<{ doc_id: string; count: number; first_pos: number; last_pos: number }>;
  }

  mentionsForDoc(docId: string): Array<{ entity_id: string; count: number }> {
    return this.db
      .prepare('SELECT entity_id, count FROM mention WHERE doc_id = ? ORDER BY count DESC')
      .all(docId) as Array<{ entity_id: string; count: number }>;
  }
```

### 出场识别服务（完整）

**文件**：`server/services/mention.ts`

```typescript
/**
 * 出场识别 —— 设定库条目在正文中的出现位置（架构 7.4）。
 *
 * 算法：多模式匹配，长名优先。
 *   把所有条目的 name + aliases 建成「首字符 → 模式列表」的桶表，每个桶内按长度降序。
 *   扫描正文时逐字符查桶，命中就用最长的那条并跳过整段。
 *   这解决了两个真实问题：
 *     - 「林远山」不会被拆成「林远」+「山」（长优先，贪心前进）
 *     - 「林远」和「林远山」两个条目同时存在时，各自拿到该拿的命中
 *
 * 为什么不用 Aho–Corasick：条目数量在几百量级，正文按场景切分后每次只扫几千字，
 * 首字符分桶已经足够快（实测 3000 字场景 < 1ms），而 AC 的失败指针实现复杂度不划算。
 *
 * 为什么不用全文索引：搜索索引是「查关键词」，出场识别是「给定词表找全部位置」，
 * 方向相反；用索引反而要发起几百次查询。
 */

import type { Entity } from '../../shared/types.ts';
import type { ProjectContext } from '../context.ts';
import type { JobContext } from '../util/job-queue.ts';
import type { IndexMention } from '../db/index.ts';
import { Mutex } from './common.ts';

interface Pattern {
  /** 匹配文本（名称或别名） */
  text: string;
  /** 该文本对应的全部条目（同名不同人时会有多个，各自都算命中） */
  entityIds: string[];
}

/** 扫描结果：entityId → docId → 命中信息 */
type MentionMap = Map<string, Map<string, { count: number; first: number; last: number }>>;

export class MentionService {
  private ctx: ProjectContext;
  private mutex = new Mutex();
  /**
   * 模式表缓存：条目变动（save/delete）时由 invalidate() 失效。
   *
   * 用独立的 loaded 标记而不是 `patterns === null` 判断是否已加载：
   * 「一个设定条目都没有」是合法且常见的初始状态，缓存下来是空数组，
   * 空数组是 truthy，用 null 判断会把「空缓存」和「未加载」混为一谈。
   */
  private patterns: Pattern[] = [];
  private patternsLoaded = false;

  constructor(ctx: ProjectContext) {
    this.ctx = ctx;
  }

  /** 条目变更后必须调用，否则扫描会用到旧的名称表 */
  invalidate(): void {
    this.patternsLoaded = false;
  }

  // ─────────────────────────── 模式表 ───────────────────────────

  /**
   * 构建模式表。
   * force=true 时忽略缓存 —— 条目变更路径（scanEntity / scanAll）必须强制刷新，
   * 否则即使调用方忘了 invalidate()，也不会用旧名字表算出脏数据。
   */
  private async buildPatterns(force = false): Promise<Pattern[]> {
    if (this.patternsLoaded && !force) return this.patterns;

    const entities = await this.ctx.store.listEntities();
    const byText = new Map<string, Pattern>();

    for (const entity of entities) {
      const names = [entity.name, ...entity.aliases]
        .map((n) => n.trim())
        .filter((n) => n.length > 0);
      for (const name of names) {
        let pattern = byText.get(name);
        if (!pattern) {
          pattern = { text: name, entityIds: [] };
          byText.set(name, pattern);
        }
        if (!pattern.entityIds.includes(entity.id)) pattern.entityIds.push(entity.id);
      }
    }

    // 长名优先：桶内排序由扫描时处理，这里只保证整体顺序稳定
    this.patterns = [...byText.values()].sort((a, b) => b.text.length - a.text.length);
    this.patternsLoaded = true;
    return this.patterns;
  }

  /** 首字符桶：每个位置只比较以该字符开头的模式 */
  private bucketize(patterns: Pattern[]): Map<string, Pattern[]> {
    const buckets = new Map<string, Pattern[]>();
    for (const pattern of patterns) {
      const first = pattern.text[0];
      const arr = buckets.get(first);
      if (arr) arr.push(pattern);
      else buckets.set(first, [pattern]);
    }
    for (const arr of buckets.values()) arr.sort((a, b) => b.text.length - a.text.length);
    return buckets;
  }

  // ─────────────────────────── 扫描 ───────────────────────────

  /**
   * 扫描单场景并写回索引。
   * indexer.indexDoc 在每次保存后调用它 —— 只扫改动的那一个场景，秒级更新。
   */
  async scanDoc(docId: string, body: string): Promise<void> {
    let patterns = await this.buildPatterns();
    if (patterns.length === 0) {
      // 缓存可能是「条目还没建」时留下的空表。保存路径不会调 invalidate，
      // 这里再强制读一次盘：代价只在「确实没有任何条目」时才发生，
      // 但如果作者刚加了条目就写正文，不重读就会漏掉这次出场。
      patterns = await this.buildPatterns(true);
    }
    if (patterns.length === 0) {
      // 确实没有任何设定条目：把该文档的历史出场记录清掉，避免删条目后残留
      await this.clearDoc(docId);
      return;
    }

    const hits = this.scanText(body, patterns);
    await this.writeDocMentions(docId, hits);
  }

  /** 场景被删除/清空后清理它的出场记录（避免反向引用里出现「幽灵章节」） */
  async clearDoc(docId: string): Promise<void> {
    const rows = this.ctx.index.mentionsForDoc(docId);
    if (rows.length === 0) return;
    this.ctx.index.transaction(() => {
      for (const row of rows) {
        this.ctx.index.replaceMentions(row.entity_id, this.withoutDoc(row.entity_id, docId));
      }
    });
  }

  /**
   * 条目变更后重扫。
   *
   * 为什么不是「只扫这个条目」：别名之间存在遮蔽关系。
   * 给 A 加一个更长的别名后，原本属于 B 的短名命中会消失；
   * 只更新 A 会让 B 的出场统计变成脏数据。因此这里做一次全书重扫，
   * 但只写回「结果确实变了」的条目，避免无谓的数据库写入。
   */
  async scanEntity(entityId: string): Promise<void> {
    const patterns = await this.buildPatterns(true);
    const tree = await this.ctx.store.readTree();
    const scenes = tree.nodes.filter((n) => n.kind === 'scene');

    const buckets = this.bucketize(patterns);
    const fresh: MentionMap = new Map();
    for (const node of scenes) {
      const scene = await this.ctx.store.readScene(node.id);
      if (!scene) continue;
      mergeInto(fresh, node.id, this.scanText(scene.body, patterns, buckets));
    }

    const dirty = new Set<string>([entityId]);
    // 其它条目的结果可能被遮蔽关系改变，一并比较
    for (const entity of await this.ctx.store.listEntities()) dirty.add(entity.id);

    for (const id of dirty) {
      const next = fresh.get(id) ?? new Map();
      if (!this.sameMentions(id, next)) this.replaceEntityMentions(id, next);
    }
  }

  /** 全量重扫（重建索引、批量导入后用） */
  async scanAll(job?: JobContext): Promise<void> {
    const patterns = await this.buildPatterns(true);
    const tree = await this.ctx.store.readTree();
    const scenes = tree.nodes.filter((n) => n.kind === 'scene');
    const total = scenes.length || 1;

    const buckets = this.bucketize(patterns);
    const fresh: MentionMap = new Map();
    for (let i = 0; i < scenes.length; i++) {
      job?.throwIfCancelled();
      const node = scenes[i];
      const scene = await this.ctx.store.readScene(node.id);
      if (!scene) continue;
      mergeInto(fresh, node.id, this.scanText(scene.body, patterns, buckets));
      if (i % 20 === 0) job?.progress((i / total) * 0.9, `识别出场 ${i + 1}/${scenes.length}`);
    }

    const entities = await this.ctx.store.listEntities();
    this.ctx.index.transaction(() => {
      for (const entity of entities) {
        this.replaceEntityMentions(entity.id, fresh.get(entity.id) ?? new Map());
      }
    });
    job?.progress(0.95, '出场索引已更新');
  }

  // ─────────────────────────── 核心匹配 ───────────────────────────

  /**
   * 在文本里找全部命中。
   * 贪心最长匹配：命中后按命中长度前进，因此不会重复计数，也不会把长名拆成短名。
   */
  private scanText(
    body: string,
    patterns: Pattern[],
    buckets: Map<string, Pattern[]> = this.bucketize(patterns),
  ): Map<string, { count: number; first: number; last: number }> {
    const result = new Map<string, { count: number; first: number; last: number }>();
    if (!body) return result;

    let pos = 0;

    while (pos < body.length) {
      const bucket = buckets.get(body[pos]);
      if (!bucket) {
        pos++;
        continue;
      }

      let matched: Pattern | null = null;
      for (const candidate of bucket) {
        if (body.startsWith(candidate.text, pos)) {
          matched = candidate;
          break; // 桶内已按长度降序，第一个命中就是最长的
        }
      }

      if (!matched) {
        pos++;
        continue;
      }

      for (const entityId of matched.entityIds) {
        const cur = result.get(entityId);
        if (cur) {
          cur.count++;
          cur.last = pos;
        } else {
          result.set(entityId, { count: 1, first: pos, last: pos });
        }
      }
      pos += matched.text.length;
    }

    return result;
  }

  // ─────────────────────────── 索引写入 ───────────────────────────

  /** 把单场景的命中写进索引：每个条目保留其它场景的记录，只替换本文档这一行 */
  private async writeDocMentions(
    docId: string,
    hits: Map<string, { count: number; first: number; last: number }>,
  ): Promise<void> {
    const affected = new Set<string>(hits.keys());
    // 之前有、这次没有的条目：需要删掉本文档那一行
    for (const m of this.ctx.index.mentionsForDoc(docId)) affected.add(m.entity_id);

    this.ctx.index.transaction(() => {
      for (const entityId of affected) {
        const next = this.withoutDoc(entityId, docId);
        const hit = hits.get(entityId);
        if (hit) {
          next.push({
            entity_id: entityId,
            doc_id: docId,
            count: hit.count,
            first_pos: hit.first,
            last_pos: hit.last,
          });
        }
        this.ctx.index.replaceMentions(entityId, next);
      }
    });
  }

  /** 取某条目除指定文档外的全部出场记录（replaceMentions 是整体替换，必须先读后写） */
  private withoutDoc(entityId: string, docId: string): IndexMention[] {
    return this.ctx.index
      .mentionsForEntity(entityId)
      .filter((m) => m.doc_id !== docId)
      .map((m) => ({
        entity_id: entityId,
        doc_id: m.doc_id,
        count: m.count,
        first_pos: m.first_pos,
        last_pos: m.last_pos,
      }));
  }

  private replaceEntityMentions(
    entityId: string,
    hits: Map<string, { count: number; first: number; last: number }>,
  ): void {
    const list: IndexMention[] = [];
    for (const [docId, hit] of hits) {
      list.push({
        entity_id: entityId,
        doc_id: docId,
        count: hit.count,
        first_pos: hit.first,
        last_pos: hit.last,
      });
    }
    this.ctx.index.replaceMentions(entityId, list);
  }

  private sameMentions(
    entityId: string,
    next: Map<string, { count: number; first: number; last: number }>,
  ): boolean {
    const current = this.ctx.index.mentionsForEntity(entityId);
    if (current.length !== next.size) return false;
    for (const m of current) {
      const hit = next.get(m.doc_id);
      if (!hit) return false;
      if (hit.count !== m.count || hit.first !== m.first_pos || hit.last !== m.last_pos) return false;
    }
    return true;
  }

  /**
   * 补全 Mention 展示信息（实体名、文档名、路径）。
   * 索引里只存 id 与位置，展示用的名字从文件读 —— 这样改名后反向引用立刻正确，
   * 不需要等索引重建。
   */
  async describeMentions(entity: Entity): Promise<
    Array<{
      doc_id: string;
      doc_title: string;
      count: number;
      first_pos: number;
      last_pos: number;
    }>
  > {
    const tree = await this.ctx.store.readTree();
    const titleById = new Map(tree.nodes.map((n) => [n.id, n.title]));
    const orderById = new Map(tree.nodes.map((n) => [n.id, n]));
    const rows = this.ctx.index.mentionsForEntity(entity.id);
    return rows
      .map((m) => ({
        doc_id: m.doc_id,
        doc_title: titleById.get(m.doc_id) ?? m.doc_id,
        count: m.count,
        first_pos: m.first_pos,
        last_pos: m.last_pos,
      }))
      .sort((a, b) => {
        const na = orderById.get(a.doc_id);
        const nb = orderById.get(b.doc_id);
        // 按文档在书里的先后排，而不是按 id 字典序
        return (na?.order ?? 0) - (nb?.order ?? 0);
      });
  }
}

// ─────────────────────────── 内部工具 ───────────────────────────

function mergeInto(
  target: MentionMap,
  docId: string,
  hits: Map<string, { count: number; first: number; last: number }>,
): void {
  for (const [entityId, hit] of hits) {
    let docs = target.get(entityId);
    if (!docs) {
      docs = new Map();
      target.set(entityId, docs);
    }
    const cur = docs.get(docId);
    if (cur) {
      cur.count += hit.count;
      cur.last = Math.max(cur.last, hit.last);
      cur.first = Math.min(cur.first, hit.first);
    } else {
      docs.set(docId, { ...hit });
    }
  }
}
```

## 4. 删除与回收站

### 设计说明

删除**永不直接擦除**。流程是「移动到 .studio/trash/ → 记元数据 → 可恢复」，
只有用户明确点「彻底删除」才真正移除。

这里修过一个会导致数据"消失"的真实 bug：最初删除章时只记录「把目录移走」，
恢复时统一丢回 `manuscript/` 根目录。结果章变成了孤立目录，它下面的场景
**整片从文档树里消失**——正文还在磁盘上，但作者看不到。连锁影响了搜索、导出、统计。

修复方式：删除时把「被移动项的原父目录」记进 `_trash.json` 的 `parent_rel`，
恢复时回原处；父目录若也不存在了，才退回 manuscript 根。

```typescript
  /** 删除节点到回收站（返回被删节点列表） */
  async deleteNode(docId: string): Promise<DocNode[]> {
    return this.write(async () => {
      const tree = await this.readTree();
      const node = tree.nodes.find((n) => n.id === docId);
      if (!node || !node.file) throw new Error(`节点不存在: ${docId}`);

      // 收集整棵子树
      const doomed: DocNode[] = [node];
      if (node.kind !== 'scene') {
        const collect = (parentId: string) => {
          for (const child of tree.nodes.filter((n) => n.parent_id === parentId)) {
            doomed.push(child);
            if (child.kind !== 'scene') collect(child.id);
          }
        };
        collect(node.id);
      }

      const abs = path.join(this.root, node.file);
      const destDir = path.join(this.paths.trash, `${Date.now()}-${docId}`);
      await ensureDir(destDir);

      // 恢复必须回到原来的父目录，否则章会被丢到 manuscript 根下变成孤立目录，
      // 它下面的场景会整片从文档树里消失（正文没丢，但作者看不到）。
      // 所以这里把「被移动项」的原父目录相对路径一并记下来。
      const movingPath = node.kind === 'scene' ? abs : path.dirname(abs);
      const parentRel = path.relative(this.root, path.dirname(movingPath));

      await fs.rename(movingPath, path.join(destDir, path.basename(movingPath)));

      await writeJsonAtomic(path.join(destDir, '_trash.json'), {
        deleted_at: Date.now(),
        root_doc_id: docId,
        // 恢复锚点：被移动项的名称 + 原父目录（相对项目 root，'' 表示 manuscript 根）
        name: path.basename(movingPath),
        parent_rel: parentRel,
        nodes: doomed,
      });

      return doomed;
    });
  }

  /** 从回收站恢复（回到原父目录；父目录已不存在时退回 manuscript 根） */
  async restoreFromTrash(trashId: string): Promise<void> {
    return this.write(async () => {
      const dir = path.join(this.paths.trash, trashId);
      if (!(await exists(dir))) throw new Error('回收站条目不存在');
      const info = await readJsonSafe<{
        nodes?: DocNode[];
        name?: string;
        parent_rel?: string;
      }>(path.join(dir, '_trash.json'), {});
      const payload = (await fs.readdir(dir)).filter((n) => n !== '_trash.json');
      if (payload.length === 0) throw new Error('回收站条目为空');

      const item = info.name && payload.includes(info.name) ? info.name : payload[0];
      const src = path.join(dir, item);

      // 优先回到原父目录；父目录若也被删了，退回 manuscript 根，至少内容不会丢
      let targetDir = info.parent_rel
        ? path.join(this.root, info.parent_rel)
        : this.paths.manuscript;
      if (!(await exists(targetDir))) {
        targetDir = this.paths.manuscript;
      }

      let finalDest = path.join(targetDir, item);
      let n = 1;
      while (await exists(finalDest)) {
        finalDest = path.join(targetDir, `${n++}-${item}`);
      }
      await ensureDir(targetDir);
      await fs.rename(src, finalDest);
      await rmSafe(dir);
    });
  }

  async listTrash(): Promise<Array<{ id: string; name: string; deleted_at: number; nodes: DocNode[] }>> {
    const dirs = await fs.readdir(this.paths.trash).catch(() => [] as string[]);
    const out: Array<{ id: string; name: string; deleted_at: number; nodes: DocNode[] }> = [];
    for (const d of dirs) {
      if (d.startsWith('.')) continue;
      const info = await readJsonSafe<{ deleted_at?: number; nodes?: DocNode[] }>(
        path.join(this.paths.trash, d, '_trash.json'),
        {},
      );
      out.push({
        id: d,
        name: info.nodes?.[0]?.title ?? d,
        deleted_at: info.deleted_at ?? 0,
        nodes: info.nodes ?? [],
      });
    }
    return out.sort((a, b) => b.deleted_at - a.deleted_at);
  }
```

```typescript
      }
    }

    await safeSync(ctx);
    appCtx.broadcast({ type: 'tree_changed', project_id: ctx.id });
    return { ok: true, deleted: deleted.length };
  });
```

### 回收站路由（完整）

**文件**：`server/routes/trash.ts`

```typescript
/**
 * 回收站路由。
 *
 * GET    /api/projects/:pid/trash              → TrashItem[]
 * POST   /api/projects/:pid/trash/:id/restore  → 恢复
 * DELETE /api/projects/:pid/trash/:id          → 彻底删除（不可恢复，前端必须二次确认）
 *
 * 回收站条目 id 就是 .studio/trash/ 下的目录名，参与路径拼接，必须先挡穿越。
 */

import path from 'node:path';
import type { FastifyInstance } from 'fastify';
import type { DocNode, TrashItem } from '../../shared/types.ts';
import type { AppContextImpl } from '../app-context.ts';
import { projectOf, syncTreeToIndex } from './shared.ts';

/** 回收站保留期：30 天。到期不自动删（自动删稿是不可接受的），只在前端标红提示 */
const RETENTION_MS = 30 * 24 * 60 * 60 * 1000;

export function registerTrashRoutes(app: FastifyInstance, appCtx: AppContextImpl): void {
  app.get('/api/projects/:pid/trash', async (request) => {
    const { pid } = request.params as { pid: string };
    const ctx = await projectOf(appCtx, pid);

    const raw = await ctx.store.listTrash();
    const items: TrashItem[] = raw.map((entry) => {
      const nodes: DocNode[] = entry.nodes ?? [];
      const root = nodes[0];
      const isContainer = nodes.length > 1 || (root ? root.kind !== 'scene' : false);
      return {
        id: entry.id,
        doc_id: root?.id ?? entry.id,
        title: root?.title ?? entry.name,
        kind: root?.kind ?? 'scene',
        path: trashPathText(ctx, root),
        deleted_at: entry.deleted_at,
        expires_at: entry.deleted_at + RETENTION_MS,
        is_container: isContainer,
        child_count: Math.max(0, nodes.length - 1),
      };
    });
    return items;
  });

  app.post('/api/projects/:pid/trash/:id/restore', async (request) => {
    const { pid, id } = request.params as { pid: string; id: string };
    const ctx = await projectOf(appCtx, pid);
    assertTrashId(id);

    await ctx.store.restoreFromTrash(id);

    try {
      await syncTreeToIndex(ctx);
    } catch (err) {
      console.error(`[trash] 恢复后索引同步失败（可重建）:`, err);
    }

    appCtx.broadcast({ type: 'tree_changed', project_id: ctx.id });
    return { ok: true };
  });

  app.delete('/api/projects/:pid/trash/:id', async (request) => {
    const { pid, id } = request.params as { pid: string; id: string };
    const ctx = await projectOf(appCtx, pid);
    assertTrashId(id);

    await ctx.store.purgeTrash(id);
    return { ok: true };
  });
}

/** 回收站目录名形如 1729999999999-01JXXXX；只允许字母数字连字符 */
function assertTrashId(id: string): void {
  if (!/^[A-Za-z0-9._-]{1,200}$/.test(id) || id === '.' || id === '..') {
    throw new Error('回收站条目标识不合法');
  }
}

/** 原始父节点路径描述：从被删节点的 parent 链在索引里还原 */
function trashPathText(
  ctx: import('../context.ts').ProjectContext,
  root: DocNode | undefined,
): string {
  if (!root) return '回收站';
  const parts: string[] = [];
  let parentId = root.parent_id;
  const guard = new Set<string>();
  while (parentId && !guard.has(parentId)) {
    guard.add(parentId);
    const parent = ctx.index.getDoc(parentId);
    if (!parent) break;
    parts.unshift(parent.title);
    parentId = parent.parent_id;
  }
  parts.push(root.title);
  return parts.join(' / ');
}

/** 供 e2e/维护脚本使用：回收站真实目录 */
export function trashDirOf(root: string): string {
  return path.join(root, '.studio', 'trash');
}
```

### 设定条目的删除

设定条目走同样的回收站路径（`deleteEntity`，见第 2 节代码），
但**不会**删除正文里的任何内容——它只是把条目的 md 文件移到 trash。

### 回收站保留策略

- **不自动清理**。自动删稿是不可接受的，无论保留多久。
- 前端对超过 30 天的条目标红提示，但只提示，不删。
- 彻底删除必须二次确认，且是唯一会真正移除文件的操作。

## 5. 前端设定库页面

### 设计说明

面板宽度只有 300px（`--inspector-w`），所以用**主从视图**而不是左右分栏：
先看列表，点进去看详情，返回再选下一个。

详情页的重点不是「再读一遍人设」，而是**反向引用**——
作者真正想知道的是「林远上一次出场是哪一章、在哪一段」。

交互要点：

- 类型筛选 chips 带数量，一眼看到「角色 2 / 地点 1」
- 搜索同时匹配名称、别名、描述
- 点击反向引用条目 → 直接跳到正文对应位置（`revealInEditor`）
- 新建/编辑用内联表单，不弹窗（300px 下弹窗太重）
- 删除走二次确认，进入回收站而非直接消失

```typescript
          const saved = list.find((e) => e.name === name && e.type === typeSelect.value);
          if (saved) await openDetail(saved.id);
          else renderList();
        } catch (e) {
          toast(ctx, `保存失败：${errText(e)}`, 'error');
        } finally {
          submitBtn.disabled = false;
        }
      }

      setTimeout(() => nameInput.focus(), 0);
    }

    // ─────────────────────── 生命周期 ───────────────────────

    let lastEntitiesRef: Entity[] | null = null;
    renderList();

    return {
      update(state: AppState, changed: Set<keyof AppState>): void {
        if (!changed.has('entities')) return;
        if (state.entities === lastEntitiesRef) return;
        lastEntitiesRef = state.entities;
        // 正在编辑时不要打断输入
        const active = document.activeElement;
        if (active && (active.tagName === 'INPUT' || active.tagName === 'TEXTAREA' || active.tagName === 'SELECT')) {
          return;
        }
        if (creating) return;
        if (selectedId) {
          // 详情页保留位置，仅刷新数据源
          const fresh = state.entities.find((e) => e.id === selectedId);
          if (!fresh) {
            renderList();
          }
          return;
        }
        renderList();
      },
      destroy(): void {
        destroyed = true;
        container.textContent = '';
      },
    };
  },
};
```

### 依赖的面板公共层

设定库用到的 `panelShell` / `emptyBlock` / `iconButton` / `searchField` / `confirmDanger`

关键片段：

**文件**：`web/src/components/panels/shared.ts`（`panelShell`）

```typescript
export function panelShell(
  container: HTMLElement,
  opts: { tools?: HTMLElement[]; toolbar?: HTMLElement } = {},
): PanelShell {
  container.textContent = '';
  const root = h('div', { class: 'panel' });

  const toolbar = h('div', { class: 'panel-toolbar' });
  if (opts.toolbar) toolbar.appendChild(opts.toolbar);
  if (opts.tools?.length) {
    const actions = h('div', { class: 'panel-toolbar-actions' });
    for (const t of opts.tools) actions.appendChild(t);
    toolbar.appendChild(actions);
  }
  toolbar.hidden = !opts.toolbar && !opts.tools?.length;

  const body = h('div', { class: 'panel-body' });
  const footer = h('div', { class: 'panel-foot' });
  footer.hidden = true;

  root.append(toolbar, body, footer);
  container.appendChild(root);

  return { root, toolbar, body, footer };
}
```

### 后端接口对应

| 前端调用 | 后端路由 |
|---|---|
| `api.listEntities()` | `GET /api/projects/:pid/entities` |
| `api.getEntity(id)` | `GET /api/projects/:pid/entities/:id` |
| `api.saveEntity(input)` | `POST /api/projects/:pid/entities` |
| `api.deleteEntity(id)` | `DELETE /api/projects/:pid/entities/:id` |

`getEntity` 返回的 `EntityDetail` 里带了 `mentions`（反向引用），
由 `MentionService.describeMentions()` 从索引读 id 与位置，再从文件读标题拼装——
这样改名后反向引用立刻正确，不需要等索引重建。
